fn is_prime(n: i32) -> i32
{
	if (n <= 1) {
		return 0
	}
	if (n <= 3) {
		return 1
	}
	if (n % 2 == 0) {
		return 0
	}
	let i = 3
	while (i * i <= n) {
		if (n % i == 0) {
			return 0
		}
		i = i + 2
	}
	return 1
}

fn main(): i32
{
	print(is_prime(0))
	print(is_prime(1))
	print(is_prime(2))
	print(is_prime(3))
	print(is_prime(4))
	print(is_prime(5))
	print(is_prime(17))
	print(is_prime(100))
	return 0
}

fn gcd(a: i32, b: i32) -> i32
{
	while (b != 0) {
		let t = b
		b = a % b
		a = t
	}
	return a
}

fn lcm(a: i32, b: i32) -> i32
{
	return (a / gcd(a, b)) * b
}

fn main(): i32
{
	print(lcm(4, 6))
	print(lcm(3, 7))
	print(lcm(12, 18))
	print(lcm(1, 100))
	return 0
}

fn bubble_sort(arr: [i32], size: i32)
{
	for (i in 0 .. size - 1) {
		for (j in 0 .. size - i - 1) {
			if (arr[j] > arr[j + 1]) {
				let temp = arr[j]
				arr[j] = arr[j + 1]
				arr[j + 1] = temp
			}
		}
	}
}

fn main(): i32
{
	let arr = [64, 34, 25, 12, 22, 11, 90]
	bubble_sort(arr, 7)
	for (i in 0 .. arr.length) {
		print(arr[i])
	}
	return 0
}

fn reverse_array(arr: [i32], size: i32)
{
	let left = 0
	let right = size - 1
	while (left < right) {
		let temp = arr[left]
		arr[left] = arr[right]
		arr[right] = temp
		left = left + 1
		right = right - 1
	}
}

fn main(): i32
{
	let arr = [1, 2, 3, 4, 5]
	reverse_array(arr, 5)
	for (i in 0 .. 5) {
		print(arr[i])
	}
	return 0
}

fn collatz_steps(n: i32) -> i32
{
	let steps = 0
	let mut val = n
	while (val != 1) {
		if (val % 2 == 0) {
			val = val / 2
		}
		else {
			val = val * 3 + 1
		}
		steps = steps + 1
	}
	return steps
}

{
	print(collatz_steps(1))
	print(collatz_steps(2))
	print(collatz_steps(6))
	print(collatz_steps(27))
	return 0
}

fn sum_arr(arr: [i32]) -> i32
{
	let s = 0
	for (i in 0 .. arr.length) {
		s = s + arr[i]
	}
	return s
}

fn main(): i32
{
	let a = [1, 2, 3, 4, 5]
	print(sum_arr(a))
	let b = [10, 20, 30]
	print(sum_arr(b))
	return 0
}
